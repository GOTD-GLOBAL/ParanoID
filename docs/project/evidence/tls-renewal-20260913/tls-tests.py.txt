import datetime as dt
import hashlib

import ipaddress

import threading
from cryptography import x509
from cryptography.hazmat.primitives import hashes,serialization
from cryptography.hazmat.primitives.asymmetric import ec
from cryptography.x509.oid import NameOID,ExtendedKeyUsageOID

ROOT=Path('/home/codex/projects/paranoid')
OUT=Path('/tmp/paranoid-renewal-local-evidence')
OUT.mkdir(exist_ok=True)
key=ec.generate_private_key(ec.SECP256R1())
pub=key.public_key().public_bytes(serialization.Encoding.DER,serialization.PublicFormat.SubjectPublicKeyInfo)
pin=hashlib.sha256(pub).hexdigest()
now=dt.datetime.utcnow().replace(microsecond=0)
name=x509.Name([x509.NameAttribute(NameOID.COMMON_NAME,'Synthetic renewal test')])
def make_cert(start,end,ip='127.0.0.1'):
 return (x509.CertificateBuilder().subject_name(name).issuer_name(name).public_key(key.public_key()).serial_number(x509.random_serial_number()).not_valid_before(start).not_valid_after(end)
 .add_extension(x509.BasicConstraints(ca=False,path_length=None),True)
 .add_extension(x509.SubjectAlternativeName([x509.IPAddress(ipaddress.ip_address(ip))]),False)
 .add_extension(x509.KeyUsage(True,False,False,False,False,False,False,False,False),True)
 .add_extension(x509.ExtendedKeyUsage([ExtendedKeyUsageOID.SERVER_AUTH]),False).sign(key,hashes.SHA256()))
old=make_cert(now-dt.timedelta(days=5),now+dt.timedelta(days=85))
new=make_cert(now-dt.timedelta(minutes=5),now+dt.timedelta(days=90)-dt.timedelta(minutes=5))
expired=make_cert(now-dt.timedelta(days=91),now-dt.timedelta(days=1))
wrong_ip=make_cert(now-dt.timedelta(minutes=5),now+dt.timedelta(days=90),'127.0.0.2')
results=[]
with tempfile.TemporaryDirectory(prefix='paranoid-renewal-synthetic-') as tmp:
 tmp=Path(tmp);keyfile=tmp/'fixture.key'
 keyfile.write_bytes(key.private_bytes(serialization.Encoding.PEM,serialization.PrivateFormat.PKCS8,serialization.NoEncryption()));keyfile.chmod(0o600)
 subprocess.run(['javac','--release','8','-Xlint:-options','-d',str(OUT),str(ROOT/'clients/android/src/org/paranoid/text/PinnedTls.java'),'/tmp/RenewalTlsProbe.java'],check=True)
 class Handler(http.server.BaseHTTPRequestHandler):
  hits=0
  def do_GET(self):
   type(self).hits+=1;self.send_response(200);self.end_headers();self.wfile.write(b'{"status":"ok"}')
  def log_message(self,*args):pass
 for label,c,selected_pin,expected in [('old',old,pin,True),('renewed_same_key',new,pin,True),('rollback_old',old,pin,True),('wrong_pin',new,'0'*64,False),('expired_same_key',expired,pin,False),('wrong_ip_same_key',wrong_ip,pin,False)]:
  crt=tmp/'fixture.crt';crt.write_bytes(c.public_bytes(serialization.Encoding.PEM))
  server=http.server.HTTPServer(('127.0.0.1',0),Handler)
  ctx=ssl.SSLContext(ssl.PROTOCOL_TLS_SERVER);ctx.load_cert_chain(crt,keyfile)
  server.socket=ctx.wrap_socket(server.socket,server_side=True)
  thread=threading.Thread(target=server.serve_forever,daemon=True);thread.start()
  before=Handler.hits
  try:
   run=subprocess.run(['java','-cp',str(OUT),'RenewalTlsProbe','127.0.0.1',str(server.server_port),selected_pin],capture_output=True,text=True,timeout=15)
   assert (run.returncode==0)==expected,(label,run.stderr)
   assert Handler.hits-before==int(expected),label
   results.append({'case':label,'result':'PASS','http_requests':Handler.hits-before})
  finally:server.shutdown();thread.join();server.server_close()
assert old.public_key().public_numbers()==new.public_key().public_numbers()
assert list(old.extensions)==list(new.extensions)
report={'result':'PASS','scope':'synthetic loopback, unchanged Android PinnedTls on JVM; NOT physical device or iOS','cases':results,'same_spki':True,'same_extensions':True}
(OUT/'result.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps(report,indent=2))
