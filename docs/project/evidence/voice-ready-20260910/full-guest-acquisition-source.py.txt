def validate_uri(item):
    filename=item['filename']
    if not filename.startswith('pool/') or any(x in filename for x in ('..','?','#','%','\\')) or not item['uri'].startswith('https://archive.ubuntu.com/ubuntu/pool/') or item['uri']!='https://archive.ubuntu.com/ubuntu/'+filename:
        raise ValueError('exact canonical package URI required')

def trusted_ancestors(path, owners):
    for p in (*reversed(path.parents),path):
        m=p.lstat()
        if not stat.S_ISDIR(m.st_mode) or m.st_uid not in owners or stat.S_IMODE(m.st_mode)&0o022:
            raise ValueError('untrusted directory ancestor')

def revalidate_links(root, entries):
    for entry in entries:
        link=root/entry['path']
        destination=(root/entry['source_target'].lstrip('/')).resolve()
        if not link.is_symlink() or os.readlink(link)!=entry['guest_equivalent_relative_target'] or not destination.is_relative_to(root) or link.resolve()!=destination:
            raise ValueError('converted guest symlink changed meaning')

def tree_manifest(root):
    result={}
    for p in sorted(root.rglob('*')):
        m=p.lstat();mode=stat.S_IMODE(m.st_mode)
        if m.st_uid!=1003:raise ValueError('unowned tooling tree member')
        if stat.S_ISLNK(m.st_mode):
            target=os.readlink(p)
            if Path(target).is_absolute() or not p.resolve().is_relative_to(root):raise ValueError('tooling link escapes tree')
            item={'kind':'symlink','target':target}
        elif stat.S_ISDIR(m.st_mode):
            if mode&0o7022:raise ValueError('unsafe tooling directory mode')
            item={'kind':'directory','mode':mode}
        elif stat.S_ISREG(m.st_mode):
            if mode&0o6022:raise ValueError('unsafe tooling file mode')
            item={'kind':'file','mode':mode,'size':m.st_size,'sha256':hashlib.sha256(p.read_bytes()).hexdigest()}
        else:raise ValueError('unexpected tooling node')
        result[str(p.relative_to(root))]=item
    return result

def guest_link_target(root, member_name, target):
    member=Path(member_name)
    if member.is_absolute() or '..' in member.parts or not target.startswith('/'):
        raise ValueError('guest link must have a safe member and absolute guest target')
    parent=(root/member).parent.resolve()
    destination=(root/target.lstrip('/')).resolve()
    if not parent.is_relative_to(root) or not destination.is_relative_to(root):
        raise ValueError('guest link escapes owned root')
    return os.path.relpath(destination,parent)
